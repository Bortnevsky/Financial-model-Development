## 3.3 Расчётное ядро

```python
# Архитектура расчётного движка

class CalculationEngine:
    """
    Граф-ориентированный расчётный движок
    """

    def __init__(self, model_id: str):
        self.graph = DependencyGraph()  # NetworkX DiGraph
        self.cells = {}                  # cell_address -> Cell
        self.formulas = {}               # cell_address -> Formula

    def set_value(self, cell: str, value: float):
        """Установить значение и запустить каскадный пересчёт"""
        self.cells[cell].override_value = value
        affected = self.graph.get_descendants(cell)
        self.recalculate(affected)

    def recalculate(self, cells: List[str]):
        """Пересчитать ячейки в топологическом порядке"""
        sorted_cells = self.graph.topological_sort(cells)
        for cell in sorted_cells:
            formula = self.formulas.get(cell)
            if formula:
                self.cells[cell].calc_value = formula.evaluate(self.cells)
```
